# backend/requirements-clone.txt
# ─────────────────────────────────────────────────────────────────────────────
# Voice cloning dependencies — install AFTER requirements.txt.
# These are intentionally separated so Standard TTS users do not need PyTorch.
#
# Usage (Windows PowerShell / Command Prompt):
#   pip install -r requirements.txt
#   pip install -r requirements-clone.txt
#   python -m unidic download
#
# Environment variables to set before starting the backend:
#   $env:OPENVOICE_ENABLED        = "true"
#   $env:OPENVOICE_CHECKPOINT_DIR = "$HOME\.cache\openvoice\v2"
#   $env:OPENVOICE_DEVICE         = "cpu"           # or "cuda"
#   $env:OPENVOICE_AUTO_DOWNLOAD  = "true"          # auto-fetch from HuggingFace
#
# Linux / macOS (bash):
#   export OPENVOICE_ENABLED=true
#   export OPENVOICE_CHECKPOINT_DIR=~/.cache/openvoice/v2
#   export OPENVOICE_DEVICE=cpu
#   export OPENVOICE_AUTO_DOWNLOAD=true
#
# Checkpoints (~300 MB) auto-download on first use when OPENVOICE_AUTO_DOWNLOAD=true.
# Manual download: python backend/checkpoint_manager.py download
# ─────────────────────────────────────────────────────────────────────────────
